# Server settings for Namecheap (Apache). The deploy workflow uploads this file as .htaccess.

# Always use the secure (https) address. Offline mode only works over https.
RewriteEngine On
RewriteCond %{HTTPS} !=on
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# No folder listings.
Options -Indexes

# The offline worker must never be cached, or phones would keep an old version of the site.
<IfModule mod_headers.c>
  <Files "sw.js">
    Header set Cache-Control "no-cache"
  </Files>
</IfModule>

<IfModule mod_mime.c>
  AddType font/woff2 .woff2
  AddType application/javascript .js
</IfModule>
